# Offline Play — Backend Sync Surface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the server endpoints that let non-ranked modes (Campaign, Practice, Category, Brain Boost) be downloaded for offline play and their results synced back and re-scored authoritatively.

**Architecture:** Two provisioning reads ship answer-bearing content bundles; two idempotent sync writes replay a client's recorded *choices*, re-score them server-side against the bank (reusing the existing `score_entry`/`compute_points` and the existing campaign settlement path), and write coins/gems/progress exactly once. Answers on the device drive only the local reveal; the server remains the sole scorer. See `docs/superpowers/specs/2026-07-11-offline-play-design.md`.

**Tech Stack:** FastAPI (async), SQLAlchemy 2.x async, Alembic, PostgreSQL, pytest/httpx. `uv` for all commands, run from `backend/`.

**Contract shared with the frontend plan (`2026-07-11-offline-play-frontend.md`):**

```
OfflineRound (in every bundle):
  { question_id: str,
    idx: int,                        # position within the level/session
    client_spec: {prompt, options[], category, icon, time_limit_ms},  # options SHUFFLED
    option_source_index: int[],      # option_source_index[shuffledPos] = original bank index
    correct_index: int,              # shuffled position of the correct option (local reveal only)
    explanation: str | null }

Sync item (client → server): { question_id: str, selected_source_index: int, elapsed_ms: int }
  selected_source_index is the ORIGINAL (pre-shuffle) bank option index the player picked.
  Server scores: correct = (selected_source_index == bank[question_id].payload.correctIndex)
```

**Prerequisite for running tests (per CLAUDE.md):** `uv run alembic upgrade head` against the test DB first; the harness uses the existing schema. Commands assume cwd = `backend/`.

```python
"""Build answer-bearing offline content bundles for non-ranked modes.

An OfflineRound carries the shuffled client_spec (for play) PLUS the local-reveal answer and a
shuffled->original option map so the client can record its choice as a STABLE original index. The
server re-scores that original index against the bank on sync (see services/offline_sync.py); the
device answer is reveal-only, never the scoring authority.
"""

from __future__ import annotations

import hashlib
import json
from random import Random
from typing import Any

from app.modules.trivia import TRIVIA_TIME_LIMIT_MS


def _shuffle_order(shuffle_seed: int, question_id: str, n: int) -> list[int]:
    """The SAME permutation app.modules.trivia.trivia_spec uses: order[j] = original index at slot j."""
    order = list(range(n))
    Random(f"{shuffle_seed}:{question_id}").shuffle(order)
    return order


def build_offline_round(q: dict[str, Any], *, idx: int, shuffle_seed: int) -> dict[str, Any]:
    payload = q["payload"]
    options: list[str] = payload["options"]
    correct = int(payload["correctIndex"])
    order = _shuffle_order(shuffle_seed, str(q["id"]), len(options))
    shuffled = [options[i] for i in order]
    correct_index = order.index(correct)  # shuffled slot holding the correct option
    return {
        "question_id": str(q["id"]),
        "idx": idx,
        "client_spec": {
            "prompt": payload["prompt"],
            "options": shuffled,
            "category": q["category"],
            "icon": q["icon"],
            "time_limit_ms": TRIVIA_TIME_LIMIT_MS,
        },
        "option_source_index": order,          # order[slot] = original index
        "correct_index": correct_index,        # shuffled slot of correct option (reveal only)
        "explanation": q.get("explanation"),
    }


def compute_bank_version(bank: list[dict[str, Any]]) -> str:
    """A content hash: changes whenever any served question's prompt/options/correctIndex changes.
    Cheap at our bank sizes; only computed on a bundle/pool fetch. Sorted by id for order-stability."""
    material = [
        [str(q["id"]), q["payload"]["prompt"], q["payload"]["options"], int(q["payload"]["correctIndex"])]
        for q in sorted(bank, key=lambda q: str(q["id"]))
    ]
    digest = hashlib.md5(json.dumps(material, separators=(",", ":")).encode()).hexdigest()
    return digest[:16]
```

```python
from sqlalchemy.ext.asyncio import AsyncSession
from content.loader import fetch_bank


OFFLINE_POOL_PER_CATEGORY = 40   # bounded per-category question pool for offline practice/category
CAMPAIGN_LOOKAHEAD_LEVELS = 3    # cache unlocked + this many locked levels ahead


async def build_practice_pool(session: AsyncSession, *, category: str | None, limit: int = OFFLINE_POOL_PER_CATEGORY) -> dict[str, Any]:
    bank = await fetch_bank(session, "trivia", category)
    chosen = bank[:limit]
    rounds = [build_offline_round(q, idx=i, shuffle_seed=hash(str(q["id"])) & 0x7FFFFFFF) for i, q in enumerate(chosen)]
    return {"category": category, "bank_version": compute_bank_version(bank), "questions": rounds}


async def build_campaign_bundle(session: AsyncSession, user_id, *, lookahead: int = CAMPAIGN_LOOKAHEAD_LEVELS) -> dict[str, Any]:
    """Build offline rounds for every unlocked level + `lookahead` locked levels ahead, per world.

    Uses the SAME manifest resolution + seed as online campaign start so an offline run scores
    identically on sync. `seed` per level is deterministic from (user_id, world, level) so the
    server can rebuild the identical shuffle if needed and re-downloads are stable.
    """
    from app.services import campaign as camp
    bank = await fetch_bank(session, "trivia")
    version = compute_bank_version(bank)
    levels_out: list[dict[str, Any]] = []
    for world, level in await camp.offline_cacheable_levels(session, user_id, lookahead=lookahead):
        seed = camp.campaign_level_seed(user_id, world.name if hasattr(world, "name") else world, level.level_number)
        rounds = [
            build_offline_round(q, idx=i, shuffle_seed=seed)
            for i, q in enumerate(camp.authored_bank_questions(level, bank))
        ]
        levels_out.append({
            "world": world.name if hasattr(world, "name") else world,
            "level": level.level_number,
            "title": level.title,
            "is_boss": level.is_boss,
            "seed": seed,
            "rounds": rounds,
        })
    return {"bank_version": version, "levels": levels_out}
```

```python
from __future__ import annotations

from typing import Any

from app.services.scoring import ScoredRound, score_entry


def score_offline_items(
    items: list[dict[str, Any]], bank_by_id: dict[str, dict[str, Any]]
) -> tuple[list[ScoredRound], int, int]:
    """Re-score recorded offline choices with the SAME engine online uses.

    Each item's selected_source_index is the ORIGINAL bank option index; we build a server_answer
    whose correctIndex is the bank's original correctIndex and a submission whose `choice` is that
    same original index, so trivia.score()'s `choice == correctIndex` comparison is correct and
    streak/points come out identical to an online run. Order of `items` defines round order (streak).
    """
    answers: list[tuple[int, str, dict[str, Any]]] = []
    submissions: dict[int, dict[str, Any]] = {}
    for idx, it in enumerate(items):
        q = bank_by_id[it["question_id"]]
        answers.append((idx, "trivia", {
            "correctIndex": int(q["payload"]["correctIndex"]),
            "question_id": it["question_id"],
            "difficulty": q.get("difficulty"),
        }))
        submissions[idx] = {"choice": int(it["selected_source_index"]), "elapsed_ms": int(it["elapsed_ms"])}
    scored, _total_points = score_entry(answers, submissions)
    correct = sum(1 for s in scored if s.correct and s.valid)
    return scored, correct, len(scored)
```

```python
from app.services.campaign import CampaignCompletion, complete_campaign_level
from app.services.offline_content import build_offline_round
from content.loader import fetch_bank

SUBMITTED = "submitted"


async def sync_campaign_offline(
    session: AsyncSession, user_id: uuid.UUID, world: str, level_number: int,
    client_id: str, items: list[dict], now: datetime | None = None,
) -> CampaignCompletion:
    now = now or datetime.now(timezone.utc)

    # 1. Idempotency: already synced → re-run complete() (idempotent) and return its recorded result.
    existing = (await session.execute(
        select(Entry).where(Entry.offline_client_id == client_id, Entry.user_id == user_id)
    )).scalar_one_or_none()
    if existing is not None:
        return await complete_campaign_level(session, existing.id, user_id, now=now)

    # 2. Resolve the level + validate the submitted questions are exactly its authored set.
    level = await camp.get_unlocked_level_or_raise(session, user_id, world, level_number, now=now)
    bank = await fetch_bank(session, "trivia")
    authored = camp.authored_bank_questions(level, bank)
    authored_ids = [str(q["id"]) for q in authored]
    submitted_ids = [it["question_id"] for it in items]
    if sorted(submitted_ids) != sorted(authored_ids):
        raise camp.CampaignError("offline items do not match the level's authored questions")

    # 3. Score in AUTHORED order (stable streak), regardless of client item order.
    by_id = {str(q["id"]): q for q in authored}
    ordered_items = [next(it for it in items if it["question_id"] == qid) for qid in authored_ids]
    scored, _correct, _total = score_offline_items(ordered_items, by_id)

    # 4. Persist a finished entry that looks exactly like an online completion.
    seed = camp.campaign_level_seed(user_id, world, level_number)
    round_set = [build_offline_round(q, idx=i, shuffle_seed=seed) for i, q in enumerate(authored)]
    entry = Entry(
        user_id=user_id, seed=seed, round_set=round_set, is_practice=True, mode="campaign",
        started_at=now, submitted_at=now, status=SUBMITTED,
        total_score=sum(s.points for s in scored), offline_client_id=client_id,
    )
    session.add(entry)
    await session.flush()
    for i, q in enumerate(authored):
        session.add(RoundAnswer(entry_id=entry.id, idx=i, module_type="trivia",
                                server_answer={"correctIndex": int(q["payload"]["correctIndex"]),
                                               "question_id": str(q["id"]), "difficulty": q.get("difficulty")}))
    for s in scored:
        session.add(RoundResult(entry_id=entry.id, idx=s.idx, module_type=s.module_type,
                                points=s.points, correct=s.correct, time_frac=s.time_frac,
                                valid=s.valid, flags=s.flags))
    session.add(CampaignSession(entry_id=entry.id, user_id=user_id, world=world, level_number=level_number))
    await session.flush()

    # 5. Reuse the EXACT online settlement (coins/gems/progress, caps, idempotency).
    return await complete_campaign_level(session, entry.id, user_id, now=now)
```

```python
async def sync_practice_offline(
    session: AsyncSession, user_id: uuid.UUID, mode: str | None, category: str | None,
    client_id: str, items: list[dict], now: datetime | None = None,
) -> PracticeResult:
    now = now or datetime.now(timezone.utc)
    existing = (await session.execute(
        select(Entry).where(Entry.offline_client_id == client_id, Entry.user_id == user_id)
    )).scalar_one_or_none()
    if existing is not None:
        return await _practice_result_from_entry(session, existing)  # rebuild the DTO from stored rows

    bank = await fetch_bank(session, "trivia", category)
    by_id = {str(q["id"]): q for q in bank}
    for it in items:
        if it["question_id"] not in by_id:
            raise ValueError("offline practice item references an unservable question")

    entry = Entry(user_id=user_id, seed=0, round_set=[], is_practice=True, mode=(mode or "practice"),
                  started_at=now, status="in_progress", offline_client_id=client_id)
    session.add(entry)
    await session.flush()
    submissions: dict[int, dict] = {}
    for i, it in enumerate(items):
        q = by_id[it["question_id"]]
        session.add(RoundAnswer(entry_id=entry.id, idx=i, module_type="trivia",
                                server_answer={"correctIndex": int(q["payload"]["correctIndex"]),
                                               "question_id": it["question_id"], "difficulty": q.get("difficulty")}))
        submissions[i] = {"choice": int(it["selected_source_index"]), "elapsed_ms": int(it["elapsed_ms"])}
    await session.flush()

    result = await submit_practice(session, entry.id, user_id, submissions, now=now)

    # Personalization signals (best-effort parity with the online per-round path).
    streak = 0
    for i, it in enumerate(items):
        q = by_id[it["question_id"]]
        correct = int(it["selected_source_index"]) == int(q["payload"]["correctIndex"])
        before = streak
        streak = streak + 1 if correct else 0
        await record_answer_signal(session, user_id, question_id=it["question_id"],
                                   mode=(mode or "practice"), is_correct=correct,
                                   time_frac=result.rounds[i].time_frac, limit_ms=TRIVIA_TIME_LIMIT_MS,
                                   streak_before=before, streak_after=streak,
                                   session_id=entry.id, difficulty=q.get("difficulty"))
    return result
```
